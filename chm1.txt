using System;

class EulerMethod
{
    // Функция f(x, y) = dy/dx для Варианта 2
    static double F(double x, double y)
    {
        return Math.Sqrt(4 * x * x + 1) - 3 * y * y;
    }

    static void Main()
    {
        // Исходные данные Варианта 2
        double a = 2.6;
        double b = 4.6;
        double y0 = 1.8;
        double h = 0.2;

        Console.WriteLine("========================================");
        Console.WriteLine("        ИСХОДНЫЕ ДАННЫЕ (Вариант 2)     ");
        Console.WriteLine("========================================");
        Console.WriteLine($" Дифференциальное уравнение: y' = sqrt(4x^2 + 1) - 3y^2");
        Console.WriteLine($" Отрезок интегрирования: [{a}; {b}]");
        Console.WriteLine($" Начальное условие: y({a}) = {y0}");
        Console.WriteLine($" Шаг интегрирования: h = {h}");
        Console.WriteLine("========================================\n");

        Console.WriteLine("=== МЕТОД ЭЙЛЕРА (h = 0.2) ===");
        Console.WriteLine("{0,-5} | {1,-10} | {2,-10} | {3,-10}", "i", "x_i", "y_i", "f(x_i, y_i)");
        Console.WriteLine(new string('-', 48));

        double y = y0;
        int step = 0;
        for (double x = a; x <= b + h / 2; x += h)
        {
            double fVal = F(x, y);
            Console.WriteLine("{0,-5} | {1,-10:F2} | {2,-10:F5} | {3,-10:F5}", step, x, y, fVal);

            y = y + h * fVal;
            step++;
        }
    }
}
