using System;

class EulerCauchyMethod
{
    static double F(double x, double y)
    {
        return Math.Sqrt(4 * x * x + 1) - 3 * y * y;
    }

    static void Main()
    {
        // Исходные данные Варианта 2
        double a = 2.6;
        double b = 4.6;
        double y0 = 1.8;
        double h = 0.2;

        Console.WriteLine("========================================");
        Console.WriteLine("        ИСХОДНЫЕ ДАННЫЕ (Вариант 2)     ");
        Console.WriteLine("========================================");
        Console.WriteLine($" Дифференциальное уравнение: y' = sqrt(4x^2 + 1) - 3y^2");
        Console.WriteLine($" Отрезок интегрирования: [{a}; {b}]");
        Console.WriteLine($" Начальное условие: y({a}) = {y0}");
        Console.WriteLine($" Шаг интегрирования: h = {h}");
        Console.WriteLine("========================================\n");

        Console.WriteLine("=== МЕТОД ЭЙЛЕРА-КОШИ (h = 0.2) ===");
        Console.WriteLine("{0,-5} | {1,-10} | {2,-10}", "i", "x_i", "y_i");
        Console.WriteLine(new string('-', 32));

        double y = y0;
        int step = 0;
        for (double x = a; x <= b + h / 2; x += h)
        {
            Console.WriteLine("{0,-5} | {1,-10:F2} | {2,-10:F5}", step, x, y);

            if (x < b)
            {
                double k1 = F(x, y);
                double y_predict = y + h * k1;
                double k2 = F(x + h, y_predict);
                y = y + (h / 2.0) * (k1 + k2);
            }
            step++;
        }
    }
}
